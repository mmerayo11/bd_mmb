```
from vega_datasets import data  
import pandas as pd
import altair as alt

alt.data_transformers.disable_max_rows()
df = data.birdstrikes()
df['Flight_Date'] = pd.to_datetime(df['Flight_Date'], errors='coerce')
df['Year'] = df['Flight_Date'].dt.year
df_clean = df.dropna(subset=['Year']).copy()
print(f"Registros listos para graficar: {len(df_clean)}")
```

## Ejercicio 1: Superposición de capas

```
base = alt.Chart(df_clean).encode(
    x=alt.X('Year:O', title='Año'),
    y=alt.Y('count():Q', title='Número de Impactos')
)

lineas = base.mark_line(color='steelblue')
puntos = base.mark_point(color='darkred', tooltip=True)

chart1 = (lineas + puntos).properties(
    title="Evolución anual de impactos de aves",
    width=600,
    height=300
)

chart1
```

## Ejercicio 2: Concatenación para un dashboard

```
gra_a = alt.Chart(df).mark_bar().encode(
    x=alt.X('When__Time_of_day:N', title='Momento del día'),
    y=alt.Y('count():Q', title='Impactos'),
    color='When__Time_of_day:N'
).properties(
    width=250, 
    height=250,
    title="Impactos por Momento del Día"
)

gra_b = alt.Chart(df).mark_bar().encode(
    x=alt.X('When__Phase_of_flight:N', title='Fase de Vuelo'),
    y=alt.Y('count():Q', title='Impactos'),
    color='When__Phase_of_flight:N'
).properties(
    width=250, 
    height=250,
    title="Impactos por Fase de Vuelo"
)

df['Cost__Total_$'] = pd.to_numeric(df['Cost__Total_$'], errors='coerce').fillna(0)

df_top10 = df.groupby('Wildlife__Species')['Cost__Total_$'].sum().reset_index()
df_top10 = df_top10[df_top10['Cost__Total_$'] > 0]
df_top10 = df_top10.sort_values('Cost__Total_$', ascending=False).head(10)

gra_c = alt.Chart(df_top10).mark_bar().encode(
    x=alt.X('Cost__Total_$:Q', title='Coste Total ($)'), # Ya no hace falta sum()
    y=alt.Y('Wildlife__Species:N', sort='-x', title='Especie'),
    color=alt.Color('Cost__Total_$:Q', scale=alt.Scale(scheme='viridis'), legend=None)
).properties(
    width=540, 
    height=300, 
    title="Top 10 Especies más costosas"
)

dashboard = (gra_a | gra_b) & gra_c
dashboard.display()
```

## Ejercicio 4: Gráficos de repetición

```
for col in ['Cost__Repair', 'Cost__Other']:
    df[col] = pd.to_numeric(df[col], errors='coerce').fillna(0)

repeat_chart = alt.Chart(df).mark_bar().encode(
    x=alt.X(alt.repeat("column"), type='quantitative', bin=alt.Bin(maxbins=20)),
    y='count():Q',
    color='Effect__Amount_of_damage:N'
).properties(width=200, height=200).repeat(
    column=['Cost__Repair', 'Cost__Other', 'Cost__Total_$']
```
