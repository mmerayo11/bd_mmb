## Ejercicio 3

```
import json
import boto3
import urllib.parse

sqs = boto3.client('sqs')
# Asegúrate de que la URL apunte a 'ColaDeProcesamiento' como pide el ejercicio
QUEUE_URL = 'https://sqs.us-east-1.amazonaws.com/730335175717/ColaDeProcesamiento'

def lambda_handler(event, context):
    for record in event['Records']:
        nombre_archivo = urllib.parse.unquote_plus(record['s3']['object']['key'])
        size_kb = record['s3']['object']['size'] / 1024
        
        # Creamos un diccionario con los datos
        datos = {
            "nombre": nombre_archivo,
            "tamaño_kb": round(size_kb, 2)
        }
        
        # Enviamos el mensaje serializado como JSON
        sqs.send_message(
            QueueUrl=QUEUE_URL,
            MessageBody=json.dumps(datos)
        )
```

## Ejercicio 4

```
import json

def lambda_handler(event, context):
    for record in event['Records']:
        # SQS envía el cuerpo como string, hay que parsearlo
        data = json.loads(record['body'])
        prioridad = data.get('prioridad')
        mensaje = data.get('mensaje')
        
        if prioridad == 'ALTA':
            print(f"¡PROCESANDO URGENTE: {mensaje}!")
        else:
            print("Registro guardado para después.")
```

## Ejercicio 7

```
import boto3

s3 = boto3.client('s3')

def lambda_handler(event, context):
    nombre_bucket = 'TU_NOMBRE_DE_BUCKET'
    
    response = s3.list_objects_v2(Bucket=nombre_bucket)
    
    # KeyCount indica cuántos objetos hay
    num_archivos = response.get('KeyCount', 0)
    
    print(f"El bucket {nombre_bucket} tiene {num_archivos} archivos.")
```
